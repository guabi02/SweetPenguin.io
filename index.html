sweet_penguin_schema.sql 
-- Base de Datos para Pastelería Sweet Pengüin
-- Compatible con PostgreSQL y MySQL

CREATE TABLE ingredientes (
    id VARCHAR(50) PRIMARY KEY,
    nombre VARCHAR(120) NOT NULL,
    cantidad_compra DECIMAL(10, 3) NOT NULL DEFAULT 1.000,
    unidad VARCHAR(20) NOT NULL, -- 'kg', 'lt', 'unid'
    precio_paquete DECIMAL(12, 2) NOT NULL,
    actualizado_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE productos (
    id VARCHAR(50) PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    categoria VARCHAR(80) DEFAULT 'General',
    rinde_unidades INT NOT NULL DEFAULT 1,
    porcentaje_ganancia DECIMAL(6, 2) NOT NULL DEFAULT 100.00,
    notas VARCHAR(255),
    creado_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE receta_ingredientes (
    id SERIAL PRIMARY KEY,
    producto_id VARCHAR(50) NOT NULL,
    ingrediente_id VARCHAR(50) NOT NULL,
    cantidad_usada DECIMAL(10, 4) NOT NULL,
    detalle_uso VARCHAR(100), -- Ej: 'Masa 26cm', 'Relleno', 'Packaging'
    FOREIGN KEY (producto_id) REFERENCES productos(id) ON DELETE CASCADE,
    FOREIGN KEY (ingrediente_id) REFERENCES ingredientes(id) ON DELETE RESTRICT
);

-- Vista útil para consultar costos y precios sugeridos en tiempo real
CREATE VIEW vista_costeo_productos AS
SELECT 
    p.id AS producto_id,
    p.nombre AS producto,
    p.rinde_unidades,
    SUM(ri.cantidad_usada * (i.precio_paquete / i.cantidad_compra)) AS costo_total_produccion,
    p.porcentaje_ganancia,
    SUM(ri.cantidad_usada * (i.precio_paquete / i.cantidad_compra)) * (1 + p.porcentaje_ganancia / 100) AS precio_final_sugerido,
    CEIL((SUM(ri.cantidad_usada * (i.precio_paquete / i.cantidad_compra)) * (1 + p.porcentaje_ganancia / 100)) / p.rinde_unidades) AS precio_unitario_sugerido
FROM productos p
JOIN receta_ingredientes ri ON p.id = ri.producto_id
JOIN ingredientes i ON ri.ingrediente_id = i.id
GROUP BY p.id, p.nombre, p.rinde_unidades, p.porcentaje_ganancia;
